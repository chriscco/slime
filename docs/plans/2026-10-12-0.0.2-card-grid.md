# slime 0.0.2 卡片网格 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把面板从竖向文字列表换成图标卡片网格（一排最多 5 张），面板换成液态玻璃，
加菜单栏入口和设置页，切片时加平滑动画。

**Architecture:** 网格数学抠进 `Core/GridLayout.swift`（纯函数，可单测）；方向键的
「轴」语义留给 `Core/Trigger`（状态机不知道列数），「一行等于几格」由 `PanelController`
用 `GridLayout.columns` 换算；图标在 `System/AppIcons` 按 bundle id 缓存、启动时预热；
设置是 `UserDefaults` 的类型化封装，面板只在 `show()` 时读一次，不做观察者。

**Tech Stack:** Swift 6 语言模式 / AppKit / `NSGlassEffectView`(macOS 26) /
swift-testing（`import Testing`，不是 XCTest）/ SwiftPM / 零第三方依赖

**Spec:** `docs/specs/2026-10-12-slime-0.0.2-spec.md`。
**执行前先读 spec 的 §1–§4 和 §12。**

**测试基线**：开工前 `scripts/test` 是 **48 个测试全绿**。每个任务末尾给了期望的数字 ——
**数不对就说明测试没真跑**（`scripts/test` 里那段注释解释过：CLT 下漏了搜索路径，

Create `Sources/slime/Core/GridLayout.swift`:

```swift
// Core/ —— 纯逻辑层，见 Window.swift 顶部注释。
//
// 只用 Swift 基本类型：CGSize / NSRect 是 CoreGraphics 的，这一层不许碰
// （CLAUDE.md 第六条；Window.swift 用 UInt32 而不是 CGWindowID 是同一个理由）。

import Foundation

/// 卡片网格的几何（spec §5.1）。纯函数，UI 层拿它算 frame。
public struct GridLayout: Equatable, Sendable {
  public static let maxColumns = 5

  public let count: Int
  public let columns: Int
  public let rows: Int
  /// 只显示这么多排，其余靠滚动。
  public let visibleRows: Int
  public let cardWidth: Double
  public let cardHeight: Double
  public let gap: Double
  public let padding: Double
  /// 搜索框那一块的高度（含上下留白）。
  public let headerHeight: Double

  // shortcut: 网格常量（卡片尺寸 / 间隙 / 最多 3 行可见）是编译期常量，
  //           被嫌弃时再做可配置（spec §14）
  public init(
    count: Int,
    cardWidth: Double = 112,
    cardHeight: Double = 92,
    gap: Double = 12,
    padding: Double = 12,
    headerHeight: Double = 46,
    maxVisibleRows: Int = 3
  ) {
    let n = max(0, count)
    self.count = n
    self.cardWidth = cardWidth
    self.cardHeight = cardHeight
    self.gap = gap
    self.padding = padding
    self.headerHeight = headerHeight
    self.columns = min(n, Self.maxColumns)
    self.rows = (n + Self.maxColumns - 1) / Self.maxColumns
    self.visibleRows = min(rows, max(0, maxVisibleRows))
  }

  /// 网格本身的宽（不含 padding）。
  public var gridWidth: Double {
    Double(columns) * cardWidth + Double(max(0, columns - 1)) * gap
  }

  /// 网格本身的高（不含 padding 和 headerHeight），只算可见的排。
  public var gridHeight: Double {
    Double(visibleRows) * cardHeight + Double(max(0, visibleRows - 1)) * gap
  }

  public var panelWidth: Double { 2 * padding + gridWidth }
  public var panelHeight: Double { 2 * padding + headerHeight + gridHeight }

  /// 第 index 张卡在**网格坐标系**（左上角原点、y 向下）里的位置。
  /// 不含 padding / headerHeight —— 那是面板内容视图的活。
  public func origin(of index: Int) -> (x: Double, y: Double) {
    guard columns > 0 else { return (0, 0) }
    return (
      Double(index % columns) * (cardWidth + gap),
      Double(index / columns) * (cardHeight + gap)
    )
  }

  /// 位移没跨行、或只跨一排 → 动画；跨更多排 → 直接跳。
  ///
  /// 只判「首尾回绕」不够：5 列网格里 ↑ 在首排往上走会绕到末排，
  /// 那种斜穿同样难看。按行距判一条规则覆盖两种情况。
  ///
  /// shortcut: 按行距判（跨 >1 排就跳），没做「按像素距离 / 按方向」的更细的
  ///           手感调优 —— 真的觉得某些移动的动画多余时再说（spec §14）
  public func shouldAnimate(from: Int, to: Int) -> Bool {
    guard count > 0, rows > 1, columns > 0 else { return true }
    return abs((to / columns) - (from / columns)) <= 1
  }
}
```

```swift
  private var columnsPerRow: Int { 1 }

  /// 网格里 ↑↓ 走一排 = `columnsPerRow` 格（spec §6）。Tab / ←→ 恒为一格。
  private func moveSelection(axis: Axis, step: Int) {

    selection = (selection + delta + rows.count) % rows.count

    panel.content.listView.scrollSelectionToVisible()
  }
```

```swift
import AppKit

/// app 图标，按 bundle id 缓存。
///
/// 取图标是主线程 API（`NSWorkspace`），而枚举在后台线程（红线 4）。所以启动时
/// 那次枚举完成后先预热一遍（`AppIcons.warm`，从 `WindowCache.onChange` 调，
/// 那里天然在主线程），快捷键路径上只读缓存、一次都不取。
///
/// `@MainActor` 是必需的，不是装饰：`NSWorkspace` / `NSRunningApplication`
/// 都是主线程 API，而这份缓存也是可变全局状态。
@MainActor
enum AppIcons {
  private static var cache: [String: NSImage] = [:]

  /// 取不到时的兜底。SF Symbol 一定存在，不会返回 nil。
  private static let fallback: NSImage = {
    let image = NSImage(systemSymbolName: "macwindow", accessibilityDescription: nil)
    return image ?? NSImage(size: NSSize(width: 64, height: 64))
  }()

  /// 预热：把快照里每个 app 的图标先取一遍。
  static func warm(_ windows: [Window]) {
    for window in windows { _ = icon(for: window) }
  }

  static func icon(for window: Window) -> NSImage? {
    guard let key = cacheKey(for: window.pid) else { return fallback }
    if let hit = cache[key] { return hit }
    guard
      let app = NSRunningApplication(processIdentifier: window.pid),
      let url = app.bundleURL
    else { return fallback }
    // shortcut: 缓存只增不删 —— 上限是「这台机器上见过多少个 app」，几十个量级。
    //           变成问题时按 pid 消失回收。
    let image = NSWorkspace.shared.icon(forFile: url.path)
    cache[key] = image
    return image
  }

  /// bundle id 优先（同一 app 的多个窗口共用一张图标）；拿不到就退回可执行路径。
  private static func cacheKey(for pid: pid_t) -> String? {
    guard let app = NSRunningApplication(processIdentifier: pid) else { return nil }
    return app.bundleIdentifier ?? app.bundleURL?.path
  }
}
```

```swift
import AppKit

/// 设置（spec §8）。放 System/ 不放 Core/ —— 读写全局可变状态是副作用，
/// 不是纯逻辑（CLAUDE.md 第六条）。
///
/// 注入 `UserDefaults` 是为了测试不污染真的 defaults，不是为了「将来换实现」。
struct Settings {
  enum GlassStyle: String, CaseIterable {
    case regular, clear

    var glassEffectStyle: NSGlassEffectView.Style {
      self == .regular ? .regular : .clear
    }
  }

  static let cornerRadiusRange: ClosedRange<Double> = 0...40
  static let defaultCornerRadius: Double = 12

  private let defaults: UserDefaults

  init(defaults: UserDefaults = .standard) {
    self.defaults = defaults
  }

  var glassStyle: GlassStyle {
    get {
      defaults.string(forKey: Key.glassStyle).flatMap(GlassStyle.init(rawValue:)) ?? .regular
    }
    nonmutating set { defaults.set(newValue.rawValue, forKey: Key.glassStyle) }
  }

  var glassTint: NSColor? {
    get {
      guard let data = defaults.data(forKey: Key.glassTint) else { return nil }
      return try? NSKeyedUnarchiver.unarchivedObject(ofClass: NSColor.self, from: data)
    }
    nonmutating set {
      guard
        let newValue,
        let data = try? NSKeyedArchiver.archivedData(
          withRootObject: newValue, requiringSecureCoding: true)
      else {
        defaults.removeObject(forKey: Key.glassTint)
        return
      }
      defaults.set(data, forKey: Key.glassTint)
    }
  }

  /// 读和写都钳位 —— 值可能是旧版本写的、或者被人手动改过。
  var glassCornerRadius: Double {
    get {
      guard defaults.object(forKey: Key.glassCornerRadius) != nil else {
        return Self.defaultCornerRadius
      }
      return Self.clampCornerRadius(defaults.double(forKey: Key.glassCornerRadius))
    }
    nonmutating set {
      defaults.set(Self.clampCornerRadius(newValue), forKey: Key.glassCornerRadius)
    }
  }

  /// 钳位是这一层唯一的纯逻辑，也是唯一的单测点。
  static func clampCornerRadius(_ value: Double) -> Double {
    guard value.isFinite else {
      return value > 0 ? cornerRadiusRange.upperBound : cornerRadiusRange.lowerBound
    }
    return min(max(value, cornerRadiusRange.lowerBound), cornerRadiusRange.upperBound)
  }

  private enum Key {
    static let glassStyle = "glass.style"
    static let glassTint = "glass.tint"
    static let glassCornerRadius = "glass.cornerRadius"
  }
}
```

```swift
import AppKit

/// 卡片网格：图标在上、窗口名在下，一排最多 5 张（spec §5）。
///
/// 不用 `NSCollectionView`：这里不要 cell 复用、不要它自带的选中样式和布局系统，
/// 手动布局反而更可控 —— 高亮就是网格里的一个普通子视图，跟着滚动走，
/// 不需要和集合视图内部的坐标系跨层。（0.0.1 的列表也是这个理由，
/// 见 docs/modules/panel/decision.md。）
final class CardGridView: NSView {
  /// 渲染的卡片。由 PanelController 推过来（rank 结果）。
  var rows: [Window] = [] {
    didSet { rebuild() }
  }

  private(set) var layout = GridLayout(count: 0)
  private(set) var selection = 0

  private let highlight = HighlightView()
  private var cards: [CardView] = []
  /// scroll view 传下来的内容宽度。宽度为 0 时卡片的 frame 宽度是负的，
  /// 什么都画不出来 —— 0.0.1 就是栽在这里（issue #15）。
  private var contentWidth: CGFloat = 0

  override var isFlipped: Bool { true }

  init() {
    super.init(frame: .zero)
    addSubview(highlight, positioned: .below, relativeTo: nil)
  }

  required init?(coder: NSCoder) { fatalError("网格只从代码建") }

  /// document view 的宽度跟着 scroll view 的内容宽度走（高度由行数决定）。
  func setContentWidth(_ width: CGFloat) {
    guard contentWidth != width else { return }
    contentWidth = width
    applyLayout()
  }

  /// 选中移动。`animated` 只给键盘移动用 —— 刷新重建时不闪。
  func setSelection(_ index: Int, animated: Bool) {
    let previous = selection
    selection = index
    highlight.isHidden = rows.isEmpty
    guard !rows.isEmpty else { return }

    let target = highlightRect(index)
    // 跨行超过一排时直接跳：高亮斜穿整个面板看起来像坏了（spec §10）。
    if animated && layout.shouldAnimate(from: previous, to: index) {
      NSAnimationContext.runAnimationGroup { context in
        context.duration = 0.15
        context.timingFunction = CAMediaTimingFunction(name: .easeOut)
        highlight.animator().frame = target
      }
    } else {
      highlight.frame = target
    }
  }

  /// 把选中项滚进可见范围（循环到列表两端时）。
  func scrollSelectionToVisible() {
    guard !rows.isEmpty else { return }
    scrollToVisible(highlightRect(selection))
  }

  private func highlightRect(_ index: Int) -> NSRect {
    let origin = layout.origin(of: index)
    // 比卡片大一圈，卡片整个落在高亮上。
    let inset: CGFloat = 4
    return NSRect(
      x: origin.x - inset,
      y: origin.y - inset,
      width: layout.cardWidth + inset * 2,
      height: layout.cardHeight + inset * 2
    )
  }

  private func rebuild() {
    cards.forEach { $0.removeFromSuperview() }
    cards = rows.map { CardView(window: $0) }
    cards.forEach(addSubview)
    layout = GridLayout(count: rows.count)
    applyLayout()
  }

  private func applyLayout() {
    for (index, card) in cards.enumerated() {
      let origin = layout.origin(of: index)
      card.frame = NSRect(
        x: origin.x, y: origin.y,
        width: layout.cardWidth, height: layout.cardHeight)
    }
    // 宽度以 scroll view 为准（不为 0 时），高度用网格自己的行数。
    frame.size = NSSize(
      width: contentWidth > 0 ? contentWidth : layout.gridWidth,
      height: layout.gridHeight)
    highlight.frame = highlightRect(selection)
  }
}

/// 一张卡片：上面 app 图标，下面窗口名。
///
/// 标签只显示**窗口标题**，不显示 app 名 —— 图标已经说明是哪个 app 了，
/// 而 112pt 宽塞不下 `app名 — 标题`。标题为空时退回 app 名。
/// internal 而不是 private：PanelLayoutTests 要能 cast 到它。
final class CardView: NSView {
  private static let iconSide: CGFloat = 64
  private static let labelHeight: CGFloat = 16
  private static let labelGap: CGFloat = 4

  private let icon = NSImageView()
  private let label = NSTextField(labelWithString: "")

  init(window: Window) {
    super.init(frame: .zero)
    icon.image = AppIcons.icon(for: window)
    icon.imageScaling = .scaleProportionallyUpOrDown
    label.stringValue = window.title.isEmpty ? window.appName : window.title
    label.font = .systemFont(ofSize: 12)
    label.textColor = .labelColor
    label.alignment = .center
    label.lineBreakMode = .byTruncatingTail
    label.drawsBackground = false
    label.isBordered = false
    addSubview(icon)
    addSubview(label)
  }

  required init?(coder: NSCoder) { fatalError("卡片只从代码建") }

  override func layout() {
    super.layout()
    icon.frame = NSRect(
      x: (bounds.width - Self.iconSide) / 2,
      y: Self.labelGap,
      width: Self.iconSide,
      height: Self.iconSide)
    label.frame = NSRect(
      x: 6,
      y: Self.labelGap + Self.iconSide + Self.labelGap,
      width: max(0, bounds.width - 12),
      height: Self.labelHeight)
  }
}

/// 高亮条：圆角、系统强调色。动画由 `CardGridView.setSelection` 驱动。

      .cgColor
  }

  required init?(coder: NSCoder) { fatalError("高亮条只从代码建") }
}
```

Create `Sources/slime/UI/SettingsWindow.swift`:

```swift
import AppKit

/// 设置窗口（spec §8）。三个控件，就这三样。
///
/// 从 `.accessory` app 打开普通窗口必须先 `NSApp.activate()`，
/// 否则窗口会开在别的 app 后面，用户以为没反应。
@MainActor
final class SettingsWindowController {
  private var settings = Settings()
  private var window: NSWindow?

  private let styleControl = NSSegmentedControl(
    labels: ["标准", "通透"], trackingMode: .selectOne, target: nil, action: nil)
  private let tintWell = NSColorWell()
  private let radiusSlider = NSSlider(
    value: Settings.defaultCornerRadius,
    minValue: Settings.cornerRadiusRange.lowerBound,
    maxValue: Settings.cornerRadiusRange.upperBound,
    target: nil, action: nil)

  func show() {
    let window = self.window ?? makeWindow()
    self.window = window
    syncControlsFromSettings()
    NSApp.activate()
    window.makeKeyAndOrderFront(nil)
  }

  private func makeWindow() -> NSWindow {
    let window = NSWindow(
      contentRect: NSRect(x: 0, y: 0, width: 320, height: 190),
      styleMask: [.titled, .closable],
      backing: .buffered,
      defer: false
    )
    window.title = "slime 设置"
    window.isReleasedWhenClosed = false
    window.contentView = makeContentView()
    window.center()
    return window
  }

  private func makeContentView() -> NSView {
    let grid = NSGridView(views: [
      [label("玻璃样式"), styleControl],
      [label("色调"), tintWell],
      [label("圆角"), radiusSlider],
    ])
    grid.rowSpacing = 12
    grid.columnSpacing = 12
    grid.column(at: 0).xPlacement = .trailing
    grid.frame = NSRect(x: 20, y: 20, width: 280, height: 130)

    styleControl.target = self
    styleControl.action = #selector(styleChanged)
    styleControl.selectedSegment = 0

    tintWell.target = self
    tintWell.action = #selector(tintChanged)
    tintWell.colorWellStyle = .minimal

    radiusSlider.target = self
    radiusSlider.action = #selector(radiusChanged)
    radiusSlider.isContinuous = true

    let container = NSView(frame: NSRect(x: 0, y: 0, width: 320, height: 190))
    container.addSubview(grid)
    return container
  }

  private func label(_ text: String) -> NSTextField {
    NSTextField(labelWithString: text)
  }

  /// 窗口每次打开都从 settings 同步一次 —— settings 是唯一真相。
  private func syncControlsFromSettings() {
    styleControl.selectedSegment = settings.glassStyle == .regular ? 0 : 1
    if let tint = settings.glassTint { tintWell.color = tint }
    radiusSlider.doubleValue = settings.glassCornerRadius
  }

  @objc private func styleChanged() {
    settings.glassStyle = styleControl.selectedSegment == 0 ? .regular : .clear
  }

  @objc private func tintChanged() {
    // 「无色调」用滑块的极端位置表达不了，所以色板取到白就算清除。
    // shortcut: 没有显式的「无色调」按钮 —— 真嫌不够用时再加一个复选框。
    settings.glassTint = tintWell.color == .white ? nil : tintWell.color
  }

  @objc private func radiusChanged() {
    settings.glassCornerRadius = radiusSlider.doubleValue
  }
}
```
